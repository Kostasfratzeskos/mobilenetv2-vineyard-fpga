% --- packages the template does not load ---------------------------------
\usepackage{siunitx}   % \SI{202}{\mega\hertz}, and aligned numeric columns
\sisetup{group-separator={,}, detect-weight=true}

% cleveref must come after hyperref, which is why this file is \input at the
% very end of the preamble. \cref{ch:timing} prints "Chapter 8" and keeps the
% word and the number in step when chapters move - which they will.
\usepackage{cleveref}
\crefname{chapter}{Chapter}{Chapters}
\crefname{section}{Section}{Sections}
\crefname{figure}{Figure}{Figures}
\crefname{table}{Table}{Tables}

% --- where figures live ---------------------------------------------------
% `make figures' converts figures/*.svg to PDF; generated/ is for anything a
% script emits.
\graphicspath{{figures/}{generated/}}

% --- the numbers ----------------------------------------------------------
% Generated from the repository by scripts/thesis_numbers.py; regenerate with
% `make numbers'. No measured figure is typed into the text - see
% thesis/README.md for why this exists.
\input{generated/numbers}

% Figures that are not yet derivable from a committed artefact. Each macro
% there carries a comment naming its source and whether it is measured,
% derived or estimated. If a number is in neither file, it does not go in the
% text.
\input{manual_numbers}
